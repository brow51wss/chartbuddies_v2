import { getSignedUrl } from '@aws-sdk/s3-request-presigner'
import { createS3Client, getS3Config } from '../../lib/s3Client'

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const key = typeof req.query.key === 'string' ? req.query.key : ''
  const validPrefix = key.startsWith('signatures/') || key.startsWith('patient-photos/')
  if (!key || !validPrefix) {
    return res.status(400).json({ error: 'Invalid key' })
  }

  try {
    const { bucket } = getS3Config()
    const s3 = createS3Client()
    const command = new GetObjectCommand({ Bucket: bucket, Key: key })
    const url = await getSignedUrl(s3, command, { expiresIn: 3600 })

    return res.redirect(302, url)

    console.error('[signature-image] failed to generate signed URL:', err)
    return res.status(500).json({ error: 'Could not load signature image' })
  }
}

/**

export function ymdFromDateInput(raw: string | null | undefined): string {
  if (raw == null) return ''

}

  const s = ymdFromDateInput(raw)
  if (!/^\d{4}-\d{2}-\d{2}$/.test(s)) return null
  const [ys, ms, ds] = s.split('-')
  const y = parseInt(ys, 10)
  const m = parseInt(ms, 10)
  const d = parseInt(ds, 10)
  if (!y || m < 1 || m > 12 || d < 1 || d > 31) return null
  const dt = new Date(y, m - 1, d, 12, 0, 0, 0)
  if (dt.getFullYear() !== y || dt.getMonth() !== m - 1 || dt.getDate() !== d) return null
  return dt
}

### 1. `recorded_at` validation — `pages/api/rds/vitals/index.ts`
**Verdict: PASS.** A new `isValidRecordedAt()` helper is in place, called immediately after the
`patient_id`/`recorded_at` required-field check and before `getPatientHospitalId`/the INSERT — a bad value
never reaches the query. Logic is correct: a date-only string (`YYYY-MM-DD`) is routed through
`ymdFromDateInput` + `parseLocalDateFromYMD` (which already rejects invalid calendar dates like month 13 or

`isValidRecordedAt` returns `false` and the handler returns `400 { error: 'Invalid recorded_at' }` before any
DB call. A valid ISO timestamp (e.g. `2026-09-05T10:30:00Z`) passes the same branch and proceeds to the
INSERT as before. No regression to the hospital-scoping or auth checks — both still run after validation,

### 3. Global Cache-Control — `middleware.ts`
**Verdict: PASS.** `middleware.ts` exists at the project root (did not exist before this fix) with exactly:
```
export function middleware(req: NextRequest) {
  const res = NextResponse.next()
  res.headers.set('Cache-Control', 'no-store, no-cache, must-revalidate, private')
  return res
}
export const config = { matcher: '/api/:path*' }
```
This is the HIPAA no-store header requested (§164.312(e)(1)/(c)(1)) and the matcher correctly scopes it to
`/api/*` only — this is why `/dashboard` and `/auth/login` don't pick up this middleware header; they were
never supposed to. Page routes keeping Next's own default `Cache-Control` behavior (which already tends to
be `no-store, must-revalidate` for dynamic pages) is fine and expected — the HIPAA requirement is about API
responses carrying ePHI, not static/page asset caching policy. No per-file `res.setHeader` calls were added
to individual routes — confirmed this landed structurally as instructed, not file-by-file.
